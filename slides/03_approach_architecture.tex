\begin{frame}{Phương pháp tiếp cận và kiến trúc hệ thống}
\small
\begin{columns}[T,onlytextwidth]
  \column{0.51\textwidth}
  \textbf{\color{HCMUTBlue}{Phương pháp tiếp cận}}
  \begin{itemize}
    \setlength{\itemsep}{0.35em}
    \item \textbf{Mô phỏng sự kiện rời rạc:}
      mô phỏng yêu cầu của sinh viên, phiên sử dụng và vòng đời VM.
    \item \textbf{Rule-based autoscaling:}
      bật VM theo nhu cầu, khởi động trước giờ học và tắt VM khi rảnh.
    \item \textbf{Thông tin đầu vào:}
      số session, hàng chờ, trạng thái VM và lịch thực hành.
  \end{itemize}

  \vspace{0.18cm}
  \textbf{\color{HCMUTBlue}{Giả định của mô hình:}}
  \[
    \boxed{1\ \text{SV} = 1\ \text{session} = 1\ \text{VM}}
  \]
